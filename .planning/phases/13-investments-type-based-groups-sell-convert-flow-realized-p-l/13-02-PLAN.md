export interface FlowEntry { effectiveDate: string /* YYYY-MM-DD */; flowCents: bigint; flowCurrency: string }
export interface ContributionsReader { firstDate: string | null; at(date: string): Promise<bigint> }
export function makeContributionsReader(
entries: FlowEntry[],
budgetCurrency: string,
rateFor: (fromCcy: string, toCcy: string, date: string) => Promise<string>,
): ContributionsReader

- firstDate = min(effectiveDate) over entries (entries with flow 0 still count: the boundary is the first HISTORY entry, D-08), null when entries empty
- at(date) = Σ over entries with effectiveDate <= date of round_half_even(flowCents * rate(flowCcy -> budgetCcy, date)); same currency => rate "1" without calling rateFor
- rateFor memoized per (fromCcy, date) inside the reader
