export const TYPE_KEYS: UiType[] // = UI_TYPE_ORDER (11 keys, no possession)
export interface TypeBlock { uiType: UiType; holdings: HoldingDto[] }
export function typeOf(h: HoldingDto): UiType // deriveUiType(h.uiType, h.holdingType, h.isCustom)
export function buildTypeBlocks(holdings: HoldingDto[], order: string[]): TypeBlock[] - only types with >= 1 holding (D-13/D-14); holdings sorted by sortOrder; block order = saved `order` first (unknown keys ignored), then remaining types in TYPE_KEYS order
export interface TypeGroupPL { valueBudgetCents: number; unrealizedCents: number; realizedCents: number; totalPLCents: number; lifetimeCostCents: number; pct: number | null }
export function typeGroupPL(holdings: HoldingDto[], summary?: { realizedBudgetCents: string; soldCostBudgetCents: string }): TypeGroupPL - held cost per holding = Number(costInBudgetCents) when present, else derived value/(1+pl/100) (optimistic rows), 0 when no basis - unrealized = Σ(value - cost) over holdings with a basis; total = unrealized + realized; lifetimeCost = Σheld cost + soldCost; pct = lifetimeCost > 0 ? total/lifetimeCost*100 : null (D-16, never ÷ held cost)
export function portfolioPL(holdings: HoldingDto[], summary: TypeSummaryDto | undefined): TypeGroupPL & { hiddenRealizedCents: number } - Σ over ALL type keys in summary.types, including types with no holdings (D-14); hiddenRealizedCents = realized of types with no holdings
export function moveTypeBlock(order: UiType[], active: UiType, over: UiType): UiType[] // arrayMove semantics
</interfaces>
